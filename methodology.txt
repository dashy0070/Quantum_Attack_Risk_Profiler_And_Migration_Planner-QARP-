import matplotlib
matplotlib.rcParams['svg.fonttype'] = 'none'
import matplotlib.pyplot as plt
import matplotlib.patches as patches

TNR = 'Times New Roman'

def txt(ax, x, y, s, fs=11.0, bold=False, color='#0F172A', ha='center', va='center'):
    ax.text(x, y, s, fontsize=fs, color=color, ha=ha, va=va,
            fontweight='bold' if bold else 'normal', fontfamily=TNR)

def draw_pill(ax, x, y, w, h, text, bg_col, text_col='white', fs=11.0):
    pill = patches.FancyBboxPatch(
        (x, y), w, h, boxstyle='round,pad=0.005,rounding_size=0.025',
        facecolor=bg_col, edgecolor='none', zorder=3)
    ax.add_patch(pill)
    txt(ax, x + w/2, y + h/2, text, fs=fs, bold=True, color=text_col)

def draw_left_card(ax, x, y, w, h, title, subtitle, border_col, bg_col, title_col):
    card = patches.FancyBboxPatch(
        (x, y), w, h, boxstyle='round,pad=0.005,rounding_size=0.015',
        facecolor=bg_col, edgecolor=border_col, linewidth=2.0, zorder=2)
    ax.add_patch(card)
    txt(ax, x + w/2, y + h*0.65, title, fs=12.0, bold=True, color=title_col)
    txt(ax, x + w/2, y + h*0.30, subtitle, fs=10.0, bold=False, color=title_col)

def draw_right_card(ax, x, y, w, h, ch_tag, title, desc, accent_col):
    # Base card with soft shadow/border
    card = patches.FancyBboxPatch(
        (x, y), w, h, boxstyle='round,pad=0.006,rounding_size=0.015',
        facecolor='#FFFFFF', edgecolor='#E2E8F0', linewidth=1.8, zorder=2)
    ax.add_patch(card)
    
    # Left accent colored bar
    bar = patches.FancyBboxPatch(
        (x + 0.002, y + 0.008), 0.008, h - 0.016, boxstyle='round,pad=0.001,rounding_size=0.004',
        facecolor=accent_col, edgecolor='none', zorder=3)
    ax.add_patch(bar)
    
    # Chapter Tag Pill
    tag_w = 0.090
    tag_h = 0.028
    tag_x = x + 0.025
    tag_y = y + h - tag_h - 0.014
    tag = patches.FancyBboxPatch(
        (tag_x, tag_y), tag_w, tag_h, boxstyle='round,pad=0.003,rounding_size=0.008',
        facecolor='#F1F5F9', edgecolor=accent_col, linewidth=1.2, zorder=3)
    ax.add_patch(tag)
    txt(ax, tag_x + tag_w/2, tag_y + tag_h/2, ch_tag, fs=8.5, bold=True, color=accent_col)
    
    # Title Text
    txt(ax, tag_x + tag_w + 0.018, tag_y + tag_h/2, title, fs=11.5, bold=True, color='#0F172A', ha='left')
    
    # Description Text
    txt(ax, x + 0.025, y + 0.022, desc, fs=9.5, bold=False, color='#475569', ha='left')
