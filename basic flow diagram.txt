import matplotlib
matplotlib.rcParams['svg.fonttype'] = 'none'
import matplotlib.pyplot as plt
import matplotlib.patches as patches

TNR = 'Times New Roman'

def txt(ax, x, y, s, fs=11.0, bold=False, color='#0F172A', ha='center', va='center'):
    ax.text(x, y, s, fontsize=fs, color=color, ha=ha, va=va,
            fontweight='bold' if bold else 'normal', fontfamily=TNR)

def draw_stage_box(ax, x, y, w, h, step_num, title, subtext, theme_col, light_bg):
    # Main rounded card body
    card = patches.FancyBboxPatch(
        (x, y), w, h, boxstyle='round,pad=0.008,rounding_size=0.015',
        facecolor=light_bg, edgecolor=theme_col, linewidth=2.2, zorder=2)
    ax.add_patch(card)

    # Left colored vertical accent strip
    accent = patches.FancyBboxPatch(
        (x, y), 0.018, h, boxstyle='round,pad=0.0,rounding_size=0.008',
        facecolor=theme_col, edgecolor=theme_col, linewidth=0, zorder=3)
    ax.add_patch(accent)

    # Step Number Pill Badge
    pill_w = 0.080
    pill_h = 0.038
    pill_x = x + 0.032
    pill_y = y + h/2 - pill_h/2
    pill = patches.FancyBboxPatch(
        (pill_x, pill_y), pill_w, pill_h, boxstyle='round,pad=0.004,rounding_size=0.010',
        facecolor=theme_col, edgecolor='none', zorder=3)
    ax.add_patch(pill)
    txt(ax, pill_x + pill_w/2, pill_y + pill_h/2, step_num, fs=11.0, bold=True, color='white')

    # Main Stage Title & Subtitle
    txt(ax, pill_x + pill_w + 0.020, y + h*0.64, title, fs=13.5, bold=True, color='#0F172A', ha='left')
    txt(ax, pill_x + pill_w + 0.020, y + h*0.34, subtext, fs=10.5, bold=False, color='#475569', ha='left')

def draw_side_callout(ax, x, y, w, h, title, details, theme_col, light_bg):
    # Callout Container Box
    callout = patches.FancyBboxPatch(
        (x, y), w, h, boxstyle='round,pad=0.006,rounding_size=0.012',
        facecolor='#FFFFFF', edgecolor=theme_col, linewidth=1.6, zorder=2)
    ax.add_patch(callout)

    # Side callout header
    txt(ax, x + 0.018, y + h*0.68, title, fs=11.0, bold=True, color=theme_col, ha='left')
    # Full untruncated technical details
    txt(ax, x + 0.018, y + h*0.32, details, fs=9.5, bold=False, color='#334155', ha='left')

def draw_terminal_node(ax, x, y, w, h, text, subtext, border_col, bg_col, text_col):
    node = patches.FancyBboxPatch(
        (x, y), w, h, boxstyle='round,pad=0.008,rounding_size=0.020',
        facecolor=bg_col, edgecolor=border_col, linewidth=2.0, zorder=2)
    ax.add_patch(node)
    txt(ax, x + w/2, y + h*0.62, text, fs=12.0, bold=True, color=text_col)
    txt(ax, x + w/2, y + h*0.32, subtext, fs=9.5, bold=False, color='#64748B')

def generate_flow_diagram():
    fig, ax = plt.subplots(figsize=(18, 12), dpi=300)
    fig.patch.set_facecolor('#FFFFFF')
    ax.set_facecolor('#FFFFFF')
    
    ax.set_xlim(-0.02, 1.02)
    ax.set_ylim(-0.02, 1.02)
    ax.axis('off')

    # Main Diagram Title
    txt(ax, 0.50, 0.980,
        'QARP END-TO-END DATA PROCESSING & RISK PROFILING PIPELINE',
        fs=17.0, bold=True, color='#0F172A')
    txt(ax, 0.50, 0.952,
        'From Raw Banking Telemetry Ingestion to Physical Hardware Capacity Forecasting',
        fs=11.5, color='#64748B')

    # Geometry Layout
    MAIN_X = 0.050   # Main process stage left-X
    MAIN_W = 0.440   # Main process stage width
    SIDE_X = 0.540   # Side callout left-X
    SIDE_W = 0.420   # Side callout width
    
    STAGE_H = 0.110  # Height of each stage card

    # Y-coordinates for the 5 vertical stages
    Y_START = 0.810
    Y_S1    = 0.630
    Y_S2    = 0.440
    Y_S3    = 0.250
    Y_END   = 0.070

    # Colors
    C_BLUE   = '#2563EB'; BG_BLUE   = '#EFF6FF'
    C_AMBER  = '#D97706'; BG_AMBER  = '#FFFBEB'
    C_PURPLE = '#7C3AED'; BG_PURPLE = '#F5F3FF'
    C_GREEN  = '#059669'; BG_GREEN  = '#F0FDF4'
    C_SLATE  = '#475569'; BG_SLATE  = '#F8FAFC'

    # -------------------------------------------------------------
    # 1. TOP TERMINAL: Input Source Data
    # -------------------------------------------------------------
    draw_terminal_node(ax, MAIN_X + 0.050, Y_START, MAIN_W - 0.100, 0.075,
                       "[ Mock Banking Network / Production Source Logs ]",
                       "JSON REST Payloads, ISO 20022 Clearing Traces, and Passive Zeek Telemetry",
                       C_SLATE, BG_SLATE, '#0F172A')

    # -------------------------------------------------------------
    # 2. STAGE 1: Banking-Specific Protocol Parser
    # -------------------------------------------------------------
    draw_stage_box(ax, MAIN_X, Y_S1, MAIN_W, STAGE_H,
                   "STEP 1", "Banking Protocol Streaming Parser",
                   "High-speed real-time ingestion engine (1,000 to 10,000 TPS)",
                   C_BLUE, BG_BLUE)
    
    draw_side_callout(ax, SIDE_X, Y_S1 + 0.005, SIDE_W, STAGE_H - 0.010,
                      "TELEMETRY EXTRACTION & CBOM AUDIT",
                      "• Extracts URI endpoints, active ciphers & payload sizes\n• Parses ISO 20022 pacs.008/pain.001 & Zeek TLS 1.3 ssl.log\n• Generates audit-ready CycloneDX 1.6 CBOM inventory",
                      C_BLUE, BG_BLUE)

    # -------------------------------------------------------------
    # 3. STAGE 2: PQC Overhead & SLA Simulator
    # -------------------------------------------------------------
    draw_stage_box(ax, MAIN_X, Y_S2, MAIN_W, STAGE_H,
                   "STEP 2", "PQC Overhead & SLA Simulator",
                   "Transport physics modeling across standard Ethernet MTU limits",
                   C_AMBER, BG_AMBER)
    
    draw_side_callout(ax, SIDE_X, Y_S2 + 0.005, SIDE_W, STAGE_H - 0.010,
                      "CRYPTOGRAPHIC BENCHMARKS & LATENCY SIMULATION",
                      "• Injects local liboqs CPU timings (ML-KEM-768 & ML-DSA-65)\n• Models 1,500-byte Ethernet MTU packet fragmentation (splits to 4 pkts)\n• Flags high-volume payment endpoints breaching <50 ms banking SLAs",
                      C_AMBER, BG_AMBER)

    # -------------------------------------------------------------
    # 4. STAGE 3: Hardware Cost & Sizing Engine
    # -------------------------------------------------------------
    draw_stage_box(ax, MAIN_X, Y_S3, MAIN_W, STAGE_H,
                   "STEP 3", "Hardware Sizing & Capacity Engine",
                   "Translating cryptographic ops/sec loss into physical hardware clusters",
                   C_PURPLE, BG_PURPLE)
    
    draw_side_callout(ax, SIDE_X, Y_S3 + 0.005, SIDE_W, STAGE_H - 0.010,
                      "PHYSICAL HSM SIZING & BUDGET CORRECTION",
                      "• Evaluates the 85% processing throughput drop under lattice math\n• Forecasts dedicated on-premises Payment HSM appliance needs\n• Uncovers the 522.2% capacity deficit of traditional linear planning",
                      C_PURPLE, BG_PURPLE)

    # -------------------------------------------------------------
    # 5. BOTTOM TERMINAL: Executive Dashboard Deliverable
    # -------------------------------------------------------------
    draw_terminal_node(ax, MAIN_X + 0.050, Y_END, MAIN_W - 0.100, 0.075,
                       "[ Interactive Dashboard / CISO Migration Report ]",
                       "Streamlit Interface (localhost:8501) with Qubit Timelines & Procurement Forecasts",
                       C_GREEN, BG_GREEN, '#047857')

    # -------------------------------------------------------------
    # 6. CONNECTING ARROWS
    # -------------------------------------------------------------
    center_x = MAIN_X + MAIN_W / 2

    def draw_down_arrow(y_top, y_bot):
        ax.annotate('', xy=(center_x, y_bot), xytext=(center_x, y_top),
                    arrowprops=dict(arrowstyle='-|>,head_width=0.4,head_length=0.6',
                                    color='#0F172A', lw=2.2), zorder=4)

    # Main vertical data-flow arrows
    draw_down_arrow(Y_START, Y_S1 + STAGE_H)
    draw_down_arrow(Y_S1, Y_S2 + STAGE_H)
    draw_down_arrow(Y_S2, Y_S3 + STAGE_H)
    draw_down_arrow(Y_S3, Y_END + 0.075)

    # Horizontal bi-directional dashed arrows linking stages to side callouts
    def draw_side_link(y_val, color):
        arrow_y = y_val + STAGE_H / 2
        ax.plot([MAIN_X + MAIN_W, SIDE_X], [arrow_y, arrow_y],
                color=color, linestyle='--', linewidth=2.0, zorder=3)
        ax.annotate('', xy=(SIDE_X, arrow_y), xytext=(SIDE_X - 0.015, arrow_y),
                    arrowprops=dict(arrowstyle='->', color=color, lw=2.0), zorder=4)
        ax.annotate('', xy=(MAIN_X + MAIN_W, arrow_y), xytext=(MAIN_X + MAIN_W + 0.015, arrow_y),
                    arrowprops=dict(arrowstyle='->', color=color, lw=2.0), zorder=4)

    draw_side_link(Y_S1, C_BLUE)
    draw_side_link(Y_S2, C_AMBER)
    draw_side_link(Y_S3, C_PURPLE)

    # Save outputs
    png_out = r'D:\Anirban\000000_Mtech Reva\CapStone_2\qarp_pipeline_architecture.png'
    svg_out = r'D:\Anirban\000000_Mtech Reva\CapStone_2\qarp_pipeline_architecture.svg'

    plt.savefig(png_out, format='png', dpi=300, facecolor='white', bbox_inches='tight', pad_inches=0.1)
    plt.savefig(svg_out, format='svg', facecolor='white', bbox_inches='tight', pad_inches=0.1)
    print(f"Success:\n1. PNG: {png_out}\n2. SVG: {svg_out}")

if __name__ == '__main__':
    generate_flow_diagram()
